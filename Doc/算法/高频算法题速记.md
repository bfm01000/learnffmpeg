#### 模板 1：滑动窗口

```cpp
// 通用滑动窗口框架
int slidingWindow(string s) {
    unordered_map<char, int> window;
    int left = 0, right = 0;
    int res = 0;  // 根据题目定义
    while (right < s.size()) {
        char c = s[right++];
        window[c]++;
        // 当窗口需要收缩时
        while (/* 窗口不满足条件 */) {
            char d = s[left++];
            window[d]--;
        }
        // 更新结果（在窗口合法时）
        res = max(res, right - left);
    }
    return res;
}
```

**适用题目**：最长无重复子串、最小覆盖子串、长度最小的子数组、固定窗口最大值

#### 模板 4：二叉树三种递归 + 层序遍历

```cpp
// 前序（根 - 左 - 右）
void preorder(TreeNode* root) {
    if (!root) return;
    visit(root);
    preorder(root->left);
    preorder(root->right);
}
// 中序（左 - 根 - 右）—— BST 中序就是升序
void inorder(TreeNode* root) {
    if (!root) return;
    inorder(root->left);
    visit(root);
    inorder(root->right);
}
// 后序（左 - 右 - 根）—— 删除/释放、自下而上汇总
int postorder(TreeNode* root) {
    if (!root) return 0;
    int l = postorder(root->left);
    int r = postorder(root->right);
    return l + r + root->val;  // 自下而上累加
}
// 层序（BFS）
vector<vector<int>> levelOrder(TreeNode* root) {
    vector<vector<int>> res;
    if (!root) return res;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        int sz = q.size();
        vector<int> level;
        while (sz--) {
            auto node = q.front(); q.pop();
            level.push_back(node->val);
            if (node->left)  q.push(node->left);
            if (node->right) q.push(node->right);
        }
        res.push_back(level);
    }
    return res;
}
```

```cpp
void backtrack(/* 状态 */, vector<int>& path, vector<vector<int>>& res) {
    if (/* 终止条件 */) {
        res.push_back(path);
        return;
    }
    for (auto choice : /* 当前可选 */) {
        if (/* 剪枝条件 */) continue;
        path.push_back(choice);           // 做选择
        backtrack(/* 新状态 */, path, res);
        path.pop_back();                   // 撤销选择

}
```

#### 模板 7：二分查找（三种边界）

```cpp
// 1. 标准二分（找等于 target）
int binarySearch(vector<int>& nums, int target) {
    int lo = 0, hi = nums.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] == target) return mid;
        else if (nums[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}

// 2. 找左边界（第一个 >= target 的位置）
int leftBound(vector<int>& nums, int target) {
    int lo = 0, hi = nums.size();  // 注意 hi = size 不是 size-1
    while (lo < hi) {                // 注意 < 不是 <=
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] < target) lo = mid + 1;
        else                     hi = mid;
    }
    return lo;
}

// 3. 找右边界（最后一个 <= target 的位置）—— 转换为 leftBound(target+1) - 1
```

```cpp
// 找右边第一个比当前大的元素（下一个更大元素）
vector<int> nextGreater(vector<int>& nums) {
    int n = nums.size();
    vector<int> res(n, -1);
    stack<int> stk;  // 存下标，单调递减
    for (int i = 0; i < n; i++) {
        while (!stk.empty() && nums[stk.top()] < nums[i]) {
            res[stk.top()] = nums[i];
            stk.pop();
        }
        stk.push(i);
    }
    return res;
}
```

### ⛳ 项目实战话术（穿插算法 + 项目）

面试官常追问"你项目里有没有用到 XX 算法？"——准备好以下话术：

**话术 A（DP - AJB 参数自适应）：**
> "我做车载预览的 AJB 算法本质就是一个**状态机 + EMA 滤波**，可以理解为一种简化版的动态规划——当前帧的目标缓冲 `dp[i] = α × observed + (1-α) × dp[i-1]`，状态只依赖前一帧，所以用滚动变量 O(1) 空间。这种'状态只依赖前一项'的转移在 DP 题里很常见，比如爬楼梯、打家劫舍。"

**话术 B（单调队列 - 滑动窗口最大值）：**
> "做直播 ABR 时我维护过最近 60 帧队列积压数的最大值，用来判断是否触发降级。直接每次扫一遍是 O(n×k)，我用单调队列把它降到 O(n)——队列尾部弹出比当前小的元素，保证队首永远是窗口最大值。这就是 LeetCode 239 的解法。"

**话术 C（堆 - Top-K）：**
> "AJB 跟踪最近 100 帧抖动 Top-10 用最小堆。这跟 LeetCode 215（第 K 大元素）一模一样，O(n log k) 比快排全排 O(n log n) 快得多。"

**话术 D（K 路归并 - 自动剪辑多源合并）：**
> "自动剪辑合并多个视频源的关键帧索引数组就是 K 路归并问题，对应 LeetCode 23（合并 K 个有序链表）。我用 `priority_queue<pair<int64_t, int>>` 维护每个源的当前指针，O(M log K) 完成。"

### 1. 双指针与滑动窗口

#### 1.1 滑动窗口三大变种

**变种 A：最长无重复子串（LC 3）**
```cpp
int lengthOfLongestSubstring(string s) {
    unordered_map<char, int> cnt;
    int left = 0, res = 0;
    for (int right = 0; right < s.size(); right++) {
        cnt[s[right]]++;
        while (cnt[s[right]] > 1) {
            cnt[s[left++]]--;
        }
        res = max(res, right - left + 1);
    }
    return res;
}
```

**变种 B：最小覆盖子串（LC 76，困难）**
- 用 `need` 哈希表存目标字符频次，`window` 存当前窗口
- `valid` 计数满足条件的字符种类数
- 满足 `valid == need.size()` 时收缩左边界

**变种 C：固定窗口大小（LC 239 滑动窗口最大值）**
- 单调队列 + 队首维护最大值
- 队尾弹出比当前小的元素

#### 1.2 双指针经典题

**三数之和（LC 15）**：
```cpp
vector<vector<int>> threeSum(vector<int>& nums) {
    sort(nums.begin(), nums.end());
    vector<vector<int>> res;
    int n = nums.size();
    for (int i = 0; i < n - 2; i++) {
        if (i > 0 && nums[i] == nums[i-1]) continue;  // 去重
        int lo = i + 1, hi = n - 1;
        while (lo < hi) {
            int sum = nums[i] + nums[lo] + nums[hi];
            if (sum == 0) {
                res.push_back({nums[i], nums[lo], nums[hi]});
                while (lo < hi && nums[lo] == nums[lo+1]) lo++;
                while (lo < hi && nums[hi] == nums[hi-1]) hi--;
                lo++; hi--;
            } else if (sum < 0) lo++;
            else hi--;
        }
    }
    return res;
}
```

**接雨水（LC 42）双指针解法**：
```cpp
int trap(vector<int>& height) {
    int left = 0, right = height.size() - 1;
    int leftMax = 0, rightMax = 0, res = 0;
    while (left < right) {
        if (height[left] < height[right]) {
            leftMax = max(leftMax, height[left]);
            res += leftMax - height[left++];
        } else {
            rightMax = max(rightMax, height[right]);
            res += rightMax - height[right--];
        }
    }
    return res;
}
```

#### 1.3 常见陷阱
- 滑动窗口忘记**收缩条件**（while 不是 if）
- 三数之和**去重位置**搞错（i / lo / hi 三处都要去重）
- 接雨水**理解错"接多少"**：每个位置接的水 = min(左最高, 右最高) - 当前高度

#### 2.1 反转链表三件套

**LC 206 反转整个链表（必背）**：
```cpp
ListNode* reverseList(ListNode* head) {
    ListNode *prev = nullptr, *cur = head;
    while (cur) {
        ListNode *next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;

}
```

**递归版本（理解递归本质）**：
```cpp
ListNode* reverseList(ListNode* head) {
    if (!head || !head->next) return head;
    ListNode* newHead = reverseList(head->next);
    head->next->next = head;
    head->next = nullptr;
    return newHead;
}
```

**LC 92 反转链表 II（反转区间 [m, n]）**：
```cpp
ListNode* reverseBetween(ListNode* head, int m, int n) {
    ListNode dummy(0); dummy.next = head;
    ListNode* pre = &dummy;
    for (int i = 1; i < m; i++) pre = pre->next;
    ListNode* cur = pre->next;
    for (int i = 0; i < n - m; i++) {
        ListNode* next = cur->next;
        cur->next = next->next;
        next->next = pre->next;
        pre->next = next;
    }
    return dummy.next;
}
```

**LC 25 K 个一组翻转链表（字节最爱考）**：
- 每 k 个节点一组反转，不足 k 个不翻转

#### 2.2 链表环

**LC 141 判断有环（快慢指针）**：
```cpp
bool hasCycle(ListNode *head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;
    }
    return false;
}
```

**LC 142 找环入口（Floyd 算法）**：
```cpp
ListNode *detectCycle(ListNode *head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) {                    // 相遇
            ListNode *ptr = head;
            while (ptr != slow) {              // 从头和相遇点同速走
                ptr = ptr->next;
                slow = slow->next;
            }
            return ptr;                         // 相遇点即环入口
        }
    }
    return nullptr;
}
```

**证明**：设头到环入口 a 步，环入口到相遇点 b 步，环长 c。
快指针走 `a + b + n×c`，慢指针走 `a + b`。
`2(a+b) = a+b+n×c` → `a + b = n×c` → `a = n×c - b`
即从头走 a 步 = 从相遇点走 `n×c - b` 步（绕几圈再走到入口）。

#### 2.3 合并与排序

**LC 21 合并两个有序链表**：
```cpp
ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
    ListNode dummy(0), *tail = &dummy;
    while (l1 && l2) {
        if (l1->val <= l2->val) { tail->next = l1; l1 = l1->next; }
        else                     { tail->next = l2; l2 = l2->next; }
        tail = tail->next;
    }
    tail->next = l1 ? l1 : l2;
    return dummy.next;
}
```

**LC 23 合并 K 个排序链表（两种方法）**：

方法 1：分治归并（推荐，O(N log K)）
```cpp
ListNode* mergeKLists(vector<ListNode*>& lists) {
    if (lists.empty()) return nullptr;
    while (lists.size() > 1) {
        vector<ListNode*> next;
        for (int i = 0; i + 1 < lists.size(); i += 2)
            next.push_back(mergeTwoLists(lists[i], lists[i+1]));
        if (lists.size() % 2) next.push_back(lists.back());
        lists = next;
    }
    return lists[0];
}
```

方法 2：最小堆
```cpp
ListNode* mergeKLists(vector<ListNode*>& lists) {
    auto cmp = [](ListNode* a, ListNode* b) { return a->val > b->val; };
    priority_queue<ListNode*, vector<ListNode*>, decltype(cmp)> pq(cmp);
    for (auto l : lists) if (l) pq.push(l);
    ListNode dummy(0), *tail = &dummy;
    while (!pq.empty()) {
        auto node = pq.top(); pq.pop();
        tail->next = node; tail = tail->next;
        if (node->next) pq.push(node->next);
    }
    return dummy.next;
}
```

### 3. 二叉树

#### 3.1 遍历四种姿势

| 遍历方式 | 递归 | 迭代（栈/队列） |
|---|---|---|
| 前序 | 简单 | 用栈，先压右后压左 |
| 中序 | 简单 | 用栈，先一路向左 |
| 后序 | 简单 | 用栈，反转前序的"根右左" |
| 层序 | 用队列 | 用队列 + 层大小 |

**迭代中序（必会）**：
```cpp
vector<int> inorderTraversal(TreeNode* root) {
    vector<int> res;
    stack<TreeNode*> stk;
    TreeNode* cur = root;
    while (cur || !stk.empty()) {
        while (cur) {
            stk.push(cur);
            cur = cur->left;
        }
        cur = stk.top(); stk.pop();
        res.push_back(cur->val);
        cur = cur->right;
    }
    return res;
}
```

#### 4.3 经典二维 DP

**LC 1143 最长公共子序列**：
```cpp
int longestCommonSubsequence(string s, string t) {
    int n = s.size(), m = t.size();
    vector<vector<int>> dp(n+1, vector<int>(m+1, 0));
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (s[i-1] == t[j-1]) dp[i][j] = dp[i-1][j-1] + 1;
            else dp[i][j] = max(dp[i-1][j], dp[i][j-1]);

    }
    return dp[n][m];
}
```

```cpp
int minDistance(string s, string t) {

    for (int i = 0; i <= n; i++) dp[i][0] = i;
    for (int j = 0; j <= m; j++) dp[0][j] = j;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (s[i-1] == t[j-1]) dp[i][j] = dp[i-1][j-1];
            else dp[i][j] = 1 + min({dp[i-1][j], dp[i][j-1], dp[i-1][j-1]});
```

```cpp
    int start = 0, mxLen = 0;
    auto expand = [&](int l, int r) {
        while (l >= 0 && r < s.size() && s[l] == s[r]) { l--; r++; }
        if (r - l - 1 > mxLen) { start = l + 1; mxLen = r - l - 1; }
    };
    for (int i = 0; i < s.size(); i++) {
```

#### 4.4 背包问题

**0-1 背包**：每个物品选或不选
```cpp

    for (int j = capacity; j >= weight[i]; j--) {  // 逆序，防止重复选
```

**完全背包**：每个物品可重复选
```cpp

    for (int j = weight[i]; j <= capacity; j++) {  // 顺序
        dp[j] = max(dp[j], dp[j - weight[i]] + value[i]);
    }
}
```

**LC 322 零钱兑换**（完全背包变形）：
```cpp
int coinChange(vector<int>& coins, int amount) {
    vector<int> dp(amount + 1, amount + 1);
    dp[0] = 0;
    for (int i = 1; i <= amount; i++) {
        for (int c : coins) {
            if (c <= i) dp[i] = min(dp[i], dp[i - c] + 1);
        }
    }
    return dp[amount] > amount ? -1 : dp[amount];
}
```

#### 4.5 DP 解题套路

- 看到"最长/最短/最大/最小"——大概率 DP
- 看到"有多少种方案"——大概率 DP（用加法）
- 看到"是否能..."——大概率 DP（布尔型）
- 看到"必须连续"——优先想滑动窗口或一维 DP
- 看到"子序列（不要求连续）"——优先想二维 DP

### 5. 回溯

#### 5.1 三大经典回溯

**LC 46 全排列**：
```cpp
vector<vector<int>> permute(vector<int>& nums) {
    vector<vector<int>> res;
    vector<int> path;
    vector<bool> used(nums.size(), false);
    function<void()> bt = [&]() {
        if (path.size() == nums.size()) {
            res.push_back(path);
            return;
        }
        for (int i = 0; i < nums.size(); i++) {
            if (used[i]) continue;
            used[i] = true;
            path.push_back(nums[i]);
            bt();
            path.pop_back();
            used[i] = false;
        }
    };
    bt();
    return res;
}
```

**LC 78 子集**：选 or 不选
```cpp
vector<vector<int>> subsets(vector<int>& nums) {
    vector<vector<int>> res;
    vector<int> path;
    function<void(int)> bt = [&](int start) {
        res.push_back(path);
        for (int i = start; i < nums.size(); i++) {
            path.push_back(nums[i]);
            bt(i + 1);
            path.pop_back();
        }

}
```

**LC 39 组合总和**（同一元素可重复）：
```cpp
vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
    sort(candidates.begin(), candidates.end());
    vector<vector<int>> res;
    vector<int> path;
    function<void(int, int)> bt = [&](int start, int remain) {
        if (remain == 0) { res.push_back(path); return; }
        for (int i = start; i < candidates.size(); i++) {
            if (candidates[i] > remain) break;  // 剪枝
            path.push_back(candidates[i]);
            bt(i, remain - candidates[i]);       // 注意是 i 不是 i+1（可重复）
            path.pop_back();
        }
    };
    bt(0, target);
    return res;
}
```

- **去重标记**：全排列 II（含重复元素）需要 `used[i-1] == false` 跳过
- **目标值剪枝**：剩余值 < 当前选择，直接 break

**找左边界（第一个 ≥ target）**：

**找右边界（最后一个 ≤ target）**：左边界 -1 或 `upper_bound - 1`。

#### 6.2 经典变形

**LC 33 搜索旋转排序数组**：
- 关键洞察：`mid` 左右一定有一边是有序的
- 判断 `nums[lo] <= nums[mid]` 即左半有序
```cpp
int search(vector<int>& nums, int target) {
    int lo = 0, hi = nums.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (nums[mid] == target) return mid;
        if (nums[lo] <= nums[mid]) {           // 左半有序
            if (target >= nums[lo] && target < nums[mid]) hi = mid - 1;
            else lo = mid + 1;
        } else {                                 // 右半有序
            if (target > nums[mid] && target <= nums[hi]) lo = mid + 1;
            else hi = mid - 1;
        }
    }
    return -1;
}
```

#### 7.1 单调栈

**LC 42 接雨水（单调栈解法）**：
```cpp
int trap(vector<int>& height) {
    stack<int> stk;
    int res = 0;
    for (int i = 0; i < height.size(); i++) {
        while (!stk.empty() && height[stk.top()] < height[i]) {
            int bottom = stk.top(); stk.pop();
            if (stk.empty()) break;
            int w = i - stk.top() - 1;
            int h = min(height[i], height[stk.top()]) - height[bottom];
            res += w * h;
        }
        stk.push(i);
    }
    return res;
}
```

### 8. 设计题（字节必考 LRU）

#### 8.1 LC 146 LRU 缓存（必背！）

```cpp
class LRUCache {
    int cap;
    list<pair<int,int>> lst;                          // {key, value}
    unordered_map<int, list<pair<int,int>>::iterator> mp;
public:
    LRUCache(int capacity) : cap(capacity) {}
    int get(int key) {
        if (mp.find(key) == mp.end()) return -1;
        lst.splice(lst.begin(), lst, mp[key]);        // 移到头部
        return mp[key]->second;
    }
    void put(int key, int value) {
        if (mp.find(key) != mp.end()) {
            mp[key]->second = value;
            lst.splice(lst.begin(), lst, mp[key]);
            return;
        }
        if (lst.size() == cap) {
            mp.erase(lst.back().first);
            lst.pop_back();
        }
        lst.emplace_front(key, value);
        mp[key] = lst.begin();
    }
};
```

**核心数据结构**：哈希表 + 双向链表
- 哈希表：O(1) 查找
- 双向链表：O(1) 移到头部 / 删除尾部
- `list::splice` 是 O(1) 移动节点不拷贝

#### 8.2 LC 460 LFU 缓存（进阶）
- 哈希表 + 多个双向链表（每个频次一个链表）
- 维护当前最小频次 `minFreq`

#### 8.3 LC 295 数据流中位数
```cpp
class MedianFinder {
    priority_queue<int> lo;                                          // 大顶堆，存较小的一半
    priority_queue<int, vector<int>, greater<int>> hi;               // 小顶堆，存较大的一半
public:
    void addNum(int num) {
        lo.push(num);
        hi.push(lo.top()); lo.pop();
        if (lo.size() < hi.size()) {
            lo.push(hi.top()); hi.pop();
        }
    }
    double findMedian() {
        return lo.size() > hi.size() ? lo.top() : (lo.top() + hi.top()) / 2.0;
    }
};
```

```cpp
    vector<int> res;
    int top = 0, bottom = matrix.size() - 1;
    int left = 0, right = matrix[0].size() - 1;
    while (top <= bottom && left <= right) {
        for (int j = left; j <= right; j++) res.push_back(matrix[top][j]);
        top++;
        for (int i = top; i <= bottom; i++) res.push_back(matrix[i][right]);
        right--;

        }

            left++;
        }
    }

}
```

### 11. 字节面试现场技巧

#### 11.1 拿到题目的标准步骤
1. **复述题意确认理解**（5%-10% 候选人栽在没读懂题）

5. **写代码**：先讲思路再写，写完讲一遍每行在干嘛

7. **复杂度分析**：时间 + 空间

- "我先把暴力解写出来，再想优化思路"

#### 13.3 蓄水池采样（LC 382）
> 字节高频追问："从未知长度的数据流里随机取一个，每个元素等概率被选"

```cpp
int getRandom() {
    int res = 0, i = 0;
    for (ListNode* node = head; node; node = node->next, i++) {
        if (rand() % (i + 1) == 0) res = node->val;  // 第 i 个元素以 1/(i+1) 概率替换
    }
    return res;
}
```
**证明**：第 k 个元素最终保留的概率 = `1/(k+1) × (k+1)/(k+2) × ... × n/(n+1) = 1/(n+1)` 均匀。
